    Deep Dive: Mobile API (mobile_api app)

    Overview

    The Mobile API provides dedicated REST endpoints for the Edubadges companion mobile app (iOS/Android). It offers a simplified, mobile-optimized API surface with its own authentication mechanism and response formats.

    Responsibilities

    • Mobile Authentication: App-specific login and token management
    • Badge Management: Read-only access to badges for mobile users
    • Enrollment Management: LTI enrollment operations from mobile
    • Direct Award Acceptance: Mobile workflow for accepting direct awards
    • Profile Management: Mobile user profile CRUD
    • Catalog Browsing: Browse available badge classes and institutions
    • Push Notification Registration: Register device tokens for FCM

    Architecture

    Key Files

    File Purpose
    api.py Mobile API view classes and handlers

    Implementation Details

    Mobile Authentication

    class MobileAPIAuthentication(BaseAuthentication):
    """Custom authentication for mobile app API calls."""

    def authenticate(self, request):
        # Extract token from Authorization header or query param
        token = request.GET.get('token') or request.META.get('HTTP_AUTHORIZATION')

        if not token:
            return None

        # Validate token and return BadgeUser
        try:
            user = BadgeUser.objects.get(auth_token=token)
            return (user, None)
        except BadgeUser.DoesNotExist:
            return None

    Login Endpoint

    class MobileLoginView(APIView):
    """Handle mobile app login."""
    serializer_class = MobileLoginSerializer

    def post(self, request):
        serializer = self.serializer_class(data=request.data)
        serializer.is_valid(raise_exception=True)

        email = serializer.validated_data['email']
        password = serializer.validated_data['password']

        user = authenticate(username=email, password=password)
        if user is None:
            return Response({'error': 'Invalid credentials'}, status=401)

        # Generate mobile token
        token = generate_mobile_token(user)

        return Response({
            'user_id': user.entity_id,
            'token': token,
            'profile': MobileProfileSerializer(user).data,
        })

    Push Notification Registration

    class RegisterDevicesView(APIView):
    """Register device tokens for push notifications."""
    authentication_classes = [MobileAPIAuthentication]

    def post(self, request):
        user = request.user
        devices = request.data.get('devices', [])

        for device in devices:
            FCMDevice.objects.update_or_create(
                registration_id=device['token'],
                user=user,
                defaults={
                    'name': device.get('name', ''),
                    'active': True,
                }
            )

        return Response({'status': 'ok'})

    Badge Instance Endpoint

    class BadgeInstanceListView(generics.ListAPIView):
    """List badge instances for the authenticated mobile user."""
    serializer_class = MobileBadgeInstanceSerializer
    authentication_classes = [MobileAPIAuthentication]

    def get_queryset(self):
        return BadgeInstance.objects.filter(
            user=self.request.user
        ).select_related(
            'badgeclass', 'issuer', 'public_key_issuer'
        ).order_by('-issued_nano')

    Direct Award Acceptance

    class AcceptDirectAwardView(generics.GenericAPIView):
    """Accept a pending direct award."""
    serializer_class = AcceptDirectAwardSerializer
    authentication_classes = [MobileAPIAuthentication]

    def post(self, request, entity_id):
        award = get_object_or_404(DirectAward, entity_id=entity_id)

        # Verify this award belongs to the current user
        if award.identifier_value != request.user.email:
            return Response({'error': 'Not your award'}, status=403)

        # Create the badge instance
        badgeinstance = BadgeInstance.objects.create(
            badgeclass=award.badgeclass,
            issuer=award.badgeclass.issuer,
            user=request.user,
            issued_nano=int(time.time() * 1e9),
        )

        # Mark award as processed
        award.status = 'awarded'
        award.matched_user = request.user
        award.created_instance = badgeinstance
        award.save()

        # Send push notification
        send_push_notification(
            user=request.user,
            title='Badge Awarded!',
            body=f'You received {award.badgeclass.name}',
            data={'badge_instance_id': badgeinstance.entity_id},
        )

        return Response(MobileBadgeInstanceSerializer(badgeinstance).data)

    API Interface

    Authentication

    Mobile API uses a simple token-based authentication: 

    Authorization: MobileToken <token>

    Or via query parameter: 

    ?token=<token>

    Potential Improvements

    1. GraphQL for Mobile: Consider GraphQL for mobile to allow clients to fetch exactly the fields they need, reducing payload size.

    2. Offline Support: No offline caching strategy. Consider implementing a local database with sync when online.

    3. Biometric Authentication: Add fingerprint/FaceID support for app login.

    4. Badge Scanning: QR code scanning for quick badge verification in the mobile app.

    5. Deep Linking: Support deep links from push notifications to specific badge instances.
